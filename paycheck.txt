<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;

class PayComponentController extends Controller

->where('status',0)

public function store(Request $r)
{

$r->validate([

'component_name'=>'required',

'component_code'=>'required|unique:pay_components,component_code,'.$r->sno.',sno',

->where('sno',$r->sno)

else

->insert($data);

'status'=>true

namespace App\Http\Controllers;

class EmployeeSalaryMappingController extends Controller
{

public function edit($employeeSno)
{

$history=DB::table('employee_salary_mapping as m')

->leftJoin('salary_structures as s',
's.sno','=','m.salary_structure_sno')

->where('m.employee_sno',$employeeSno)

->select(

's.structure_name',
'm.effective_from',
'm.effective_to',
'm.change_reason'

)

->orderBy('m.effective_from','desc')

->get();


return response()->json([

'history'=>$history

]);

}


public function store(Request $r)
{

$r->validate([

'employee_sno'=>'required',

'salary_structure_sno'=>'required',

'effective_from'=>'required'

]);


DB::transaction(function() use($r){

// close previous active structure

DB::table('employee_salary_mapping')

->where('employee_sno',$r->employee_sno)

->whereNull('effective_to')

->update([

'effective_to'=>date('Y-m-d',strtotime($r->effective_from.' -1 day')),

'updated_by'=>auth()->id()

]);


// insert new mapping

DB::table('employee_salary_mapping')

->insert([

'employee_sno'=>$r->employee_sno,

'salary_structure_sno'=>$r->salary_structure_sno,

'effective_from'=>$r->effective_from,

'change_reason'=>$r->change_reason,

'created_by'=>auth()->id()

]);

});


return response()->json([

'status'=>true

]);

}

}

<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use DB;

class PayrollRunController extends Controller
{

public function index()
{

$runs=DB::table('payroll_runs')
->where('status',0)

->get();

return view('payroll.payroll_runs',compact('runs'));

}


public function create(Request $r)
{

DB::table('payroll_runs')
->insert([

'payroll_month'=>$r->payroll_month,

'payroll_year'=>$r->payroll_year,

'run_status'=>'DRAFT',

'created_by'=>auth()->id()

]);

return response()->json(['status'=>true]);

}


public function process($runSno)
{

DB::transaction(function() use($runSno){

$run=DB::table('payroll_runs')
->where('sno',$runSno)
->first();


$employees=DB::table('employee_payroll_profile')

->where('payroll_status','ACTIVE')

->get();


foreach($employees as $emp)
{

$employeeSno=$emp->employee_sno;


// get salary structure

$mapping=DB::table('employee_salary_mapping')

->where('employee_sno',$employeeSno)

->whereNull('effective_to')

->first();


if(!$mapping) continue;


$components=DB::table('salary_structure_components')

->where('salary_structure_sno',
$mapping->salary_structure_sno)

->get();


$gross=0;
$deduction=0;


// create payroll entry

$entryId=DB::table('payroll_entries')

->insertGetId([

'payroll_run_sno'=>$runSno,

'employee_sno'=>$employeeSno,

'gross_salary'=>0,

'total_deductions'=>0,

'net_salary'=>0,

'created_by'=>auth()->id()

]);



foreach($components as $c)
{

$amount=$c->value;


$type=DB::table('pay_components')
->where('sno',$c->component_sno)
->value('component_type');


DB::table('payroll_entry_components')

->insert([

'payroll_entry_sno'=>$entryId,

'component_sno'=>$c->component_sno,

'amount'=>$amount,

'created_by'=>auth()->id()

]);


if($type=='EARNING')
$gross+=$amount;

else
$deduction+=$amount;

}


$net=$gross-$deduction;


DB::table('payroll_entries')

->where('sno',$entryId)

->update([

'gross_salary'=>$gross,

'total_deductions'=>$deduction,

'net_salary'=>$net

]);

}


// finalize

DB::table('payroll_runs')

->where('sno',$runSno)

->update([

'run_status'=>'FINALIZED'

]);

});


return response()->json([

'status'=>true,

'message'=>'Payroll processed successfully'

]);

}

}

private function getPayslipData($entrySno)

class AttendanceService
{

public static function calculateLOP($employeeSno,$month,$year,$grossSalary)
{

$workingDays = payroll_setting('working_days_per_month');

$lateAllowed = payroll_setting('late_allowed_count');

$lopPerLate = payroll_setting('lop_per_late');


$attendance = DB::table('attendance')

->where('employee_id',$employeeSno)

->whereMonth('date',$month)

->whereYear('date',$year)

->get();


$absentCount=0;
$lateCount=0;


foreach($attendance as $a)
{

if($a->attendance_status=='A')
$absentCount++;

if($a->attendance_status=='L')
$lateCount++;

}


// calculate late LOP

$extraLate=max(0,$lateCount-$lateAllowed);

$lateLOP=$extraLate*$lopPerLate;


// total LOP days

$totalLOP=$absentCount+$lateLOP;


// daily salary

$dailySalary=$grossSalary/$workingDays;


// lop amount

$lopAmount=$dailySalary*$totalLOP;


return [

'lop_days'=>$totalLOP,

'lop_amount'=>$lopAmount

];

}

}

<?php

namespace App\Services;

use DB;

class StatutoryService
{

public static function calculatePF($grossSalary)
{

$rule = DB::table('statutory_rules')
->where('statutory_name','PF')
->where('status',0)
->orderBy('effective_from','desc')
->first();

if(!$rule) return 0;


// PF applies only up to wage limit

$pfSalary = min($grossSalary,$rule->wage_limit);

$pfAmount = ($pfSalary * $rule->employee_percentage)/100;

return round($pfAmount);

}



public static function calculateESI($grossSalary)
{

$rule = DB::table('statutory_rules')
->where('statutory_name','ESI')
->where('status',0)
->orderBy('effective_from','desc')
->first();

if(!$rule) return 0;


// only if salary <= limit

if($grossSalary > $rule->wage_limit)
return 0;


$esiAmount = ($grossSalary * $rule->employee_percentage)/100;

return round($esiAmount);

}

}

public function getPreviewData(Request $request)
{

$month = $request->month;
$year = $request->year;

$pfPercentage = payroll_setting('pf_percentage');
$esiPercentage = payroll_setting('esi_percentage');
$esiLimit = payroll_setting('esi_salary_limit');

$staff = DB::table('egc_staff_details as s')

->leftJoin('egc_staff_attendance as a', function($join) use ($month,$year){

$join->on('a.staff_id','=','s.sno')
->whereMonth('a.date',$month)
->whereYear('a.date',$year)
->where('a.status',0);

})

->where('s.status',0)

->select(

's.sno',
's.name',
's.staff_code',
's.basic_salary',
's.per_day_salary',
's.casual_leave_count_per_month',

DB::raw("SUM(CASE WHEN a.attendance='P' THEN 1 ELSE 0 END) as present_days"),

DB::raw("SUM(CASE WHEN a.attendance='A' THEN 1 ELSE 0 END) as absent_days")

)

->groupBy('s.sno')

->get();


$data = [];

foreach($staff as $row)
{

$absent = $row->absent_days ?? 0;

$allowedLeave = $row->casual_leave_count_per_month ?? 0;

$lopDays = max(0,$absent - $allowedLeave);

$lopAmount = $lopDays * $row->per_day_salary;

$pfAmount = ($row->basic_salary * $pfPercentage)/100;

$esiAmount = 0;

if($row->basic_salary <= $esiLimit)
$esiAmount = ($row->basic_salary * $esiPercentage)/100;

$netSalary = $row->basic_salary - ($lopAmount + $pfAmount + $esiAmount);

$data[] = [

'staff_id'=>$row->sno,
'name'=>$row->name,
'staff_code'=>$row->staff_code,

'basic_salary'=>$row->basic_salary,

'present_days'=>$row->present_days ?? 0,

'absent_days'=>$absent,

'lop_days'=>$lopDays,

'lop_amount'=>$lopAmount,

'pf_amount'=>$pfAmount,

'esi_amount'=>$esiAmount,

'net_salary'=>$netSalary

];

}

return response()->json($data);

}

}

public function savePayroll(Request $request)
{

$month = $request->month;
$year = $request->year;


// prevent duplicate save

$exists = DB::table('egc_payroll_runs')

->where('payroll_month',$month)
->where('payroll_year',$year)
->where('status',0)
->exists();

if($exists)
{

return response()->json([
'status'=>false,
'message'=>'Payroll already saved for this month'
]);

}


DB::beginTransaction();

try {


// create payroll run

$runId = DB::table('egc_payroll_runs')

->insertGetId([

'payroll_month'=>$month,
'payroll_year'=>$year,
'run_type'=>'MANUAL',
'created_by'=>auth()->id()

]);


// get preview data using same logic

$preview = $this->getPreviewData($request)->getData();


foreach($preview as $row)
{


$entryId = DB::table('egc_payroll_entries')

->insertGetId([

'payroll_run_sno'=>$runId,

'staff_id'=>$row->staff_id,

'basic_salary'=>$row->basic_salary,

'per_day_salary'=>$row->basic_salary,

'present_days'=>$row->present_days,

'absent_days'=>$row->absent_days,

'allowed_leave'=>0,

'lop_days'=>$row->lop_days,

'lop_amount'=>$row->lop_amount,

'pf_amount'=>$row->pf_amount,

'esi_amount'=>$row->esi_amount,

'incentive_amount'=>0,

'net_salary'=>$row->net_salary,

'created_by'=>auth()->id()

]);


// save components

DB::table('egc_payroll_entry_components')

->insert([

[
'payroll_entry_sno'=>$entryId,
'component_name'=>'Basic Salary',
'component_type'=>'EARNING',
'amount'=>$row->basic_salary
],

[
'payroll_entry_sno'=>$entryId,
'component_name'=>'LOP',
'component_type'=>'DEDUCTION',
'amount'=>$row->lop_amount
],

[
'payroll_entry_sno'=>$entryId,
'component_name'=>'PF',
'component_type'=>'DEDUCTION',
'amount'=>$row->pf_amount
],

[
'payroll_entry_sno'=>$entryId,
'component_name'=>'ESI',
'component_type'=>'DEDUCTION',
'amount'=>$row->esi_amount
]

]);

}


DB::commit();

return response()->json([
'status'=>true,
'message'=>'Payroll saved successfully'
]);

}
catch(\Exception $e)
{

DB::rollBack();

return response()->json([
'status'=>false,
'message'=>$e->getMessage()
]);

}

}

namespace App\Console\Commands;

use Illuminate\Console\Command;

class GeneratePayrollCron extends Command
{

protected $signature = 'payroll:generate';

protected $description = 'Generate Monthly Payroll Automatically';


public function handle()
{

$month = date('m');
$year = date('Y');

$this->info("Generating payroll for $month-$year");


$exists = DB::table('egc_payroll_runs')

->where('payroll_month',$month)
->where('payroll_year',$year)

->exists();


if($exists)
{

$this->info("Payroll already generated");

return;

}


$helper = new Helpers();

$pfPercentage = $helper->payroll_setting('pf_percentage');
$esiPercentage = $helper->payroll_setting('esi_percentage');
$esiLimit = $helper->payroll_setting('esi_salary_limit');

$lateAllowed = $helper->payroll_setting('late_allowed_per_month');
$lopPerLate = $helper->payroll_setting('lop_per_late');


DB::beginTransaction();

try {


$runId = DB::table('egc_payroll_runs')

->insertGetId([

'payroll_month'=>$month,
'payroll_year'=>$year,

'run_type'=>'CRON',

'created_at'=>now()

]);


$staff = DB::table('egc_staff')

->where('status',0)

->get();


foreach($staff as $s)
{


$present = DB::table('egc_staff_attendance')

->where('staff_id',$s->sno)
->whereMonth('date',$month)
->whereYear('date',$year)
->where('attendance','P')

->count();


$absent = DB::table('egc_staff_attendance')

->where('staff_id',$s->sno)
->whereMonth('date',$month)
->whereYear('date',$year)
->where('attendance','A')

->count();


$lateCount = DB::table('egc_staff_attendance')

->where('staff_id',$s->sno)
->whereMonth('date',$month)
->whereYear('date',$year)

->whereTime('in_time','>','09:30:00')

->count();


$absentLop = max(0,$absent - $s->casual_leave_count_per_month);

$lateLop = max(0,$lateCount - $lateAllowed) * $lopPerLate;

$totalLopDays = $absentLop + $lateLop;

$lopAmount = $totalLopDays * $s->per_day_salary;


$pfAmount = ($s->basic_salary * $pfPercentage)/100;

$esiAmount = 0;

if($s->basic_salary <= $esiLimit)
$esiAmount = ($s->basic_salary * $esiPercentage)/100;


$netSalary =

$s->basic_salary

- $lopAmount
- $pfAmount
- $esiAmount;


$entryId = DB::table('egc_payroll_entries')

->insertGetId([

'payroll_run_sno'=>$runId,

'staff_id'=>$s->sno,

'basic_salary'=>$s->basic_salary,

'per_day_salary'=>$s->per_day_salary,

'present_days'=>$present,

'absent_days'=>$absent,

'lop_days'=>$totalLopDays,

'lop_amount'=>$lopAmount,

'pf_amount'=>$pfAmount,

'esi_amount'=>$esiAmount,

'net_salary'=>$netSalary,

'created_at'=>now()

]);


}


DB::commit();

$this->info("Payroll generated successfully");


}
catch(\Exception $e)

DB::rollBack();

$this->error($e->getMessage());

}

 public function index(Request $request)
    {
        $helper = new \App\Helpers\Helpers();
        $page = $request->input('page', 1);
        $perpage = (int) $request->input('sorting_filter', 25);
        $offset = ($page - 1) * $perpage;
        $search_filter = $request->search_filter ?? '';
        $Document = PayrollComponentModel::where( 'status', '!=', 2 );
        if ($search_filter != '') {
            $Document->where(function ($subquery) use ($search_filter) {
                $subquery->where('component_name', 'LIKE', "%{$search_filter}%")
                    ->orWhere('component_type', 'LIKE', "%{$search_filter}%");
                    
            });
        }
        $Document=$Document->orderBy( 'sno', 'desc' )->paginate($perpage);

        if ($request->ajax()) {
            $data = $Document->map(function ($item) use ($helper) {
                return [
                    'sno' => $item->sno,
                    'status' => $item->status,
                    'component_name' => $item->component_name,
                    'component_code' => $item->component_code,
                    'is_taxable' => $item->is_taxable,
                    'is_statutory' => $item->is_statutory,
                    'is_prorated' => $item->is_prorated,
                    'component_type' => $item->component_type,
                    'show_in_payslip' => $item->show_in_payslip,
                    'item' => $item,
                    'encrypted_id' => $helper->encrypt_decrypt($item->sno, 'encrypt'),
                ];
            });

            return response()->json([
                'data' => $data,
                'current_page' => $Document->currentPage(),
                'last_page' => $Document->lastPage(),
                'total' => $Document->total(),
            ]);
        }

        return view('content.settings.payroll.new_payroll_component.new_payroll_component',[
            'Document' => $Document,
            'perpage' => $perpage,
            'search_filter' => $search_filter
        ]);

    function loadThemes(page = 1) {
        const perpage = document.getElementById('perpage').value;
        const search = document.getElementById('search_filter').value;
        const url = /settings/payroll_component?page=${page}&sorting_filter=${perpage}&search_filter=${search};

        // Show skeleton loader and clear old data before fetching new data
        isLoading = true;
        document.getElementById('list-table-body').innerHTML = ''; // Clear old data
        document.getElementById('list-table-body').insertAdjacentHTML('beforeend', skeletenRow()); // Clear old data
        $('#skeleton-loader').show(); // Show skeleton loader

        if (abortController.signal) {
            abortController.abort(); // Abort the previous request
        }
        abortController = new AbortController();

         fetch(url, { headers: { 'X-Requested-With': 'XMLHttpRequest' }, signal: abortController.signal })
            .then(res => res.json())
            .then(res => {
                // Insert new data into the table
            if(res.data.length > 0){
                res.data.forEach((item, index) => {
                    document.getElementById('list-table-body').insertAdjacentHTML('beforeend', buildRow(item, index + 1));
                });

                // Update pagination and results info
                updatePagination(res.current_page, res.last_page, res.total, perpage);

                isLoading = false;
                $('#skeleton-loader').hide();

            .catch(error => {
                if (error.name !== 'AbortError') { // Only handle abort error
                    console.error('Error loading data:', error);

    function updatePagination(currentPage, lastPage, total, perpage) {
        let paginationContainer = document.getElementById('pagination-container');
        paginationContainer.innerHTML = ''; // Clear old pagination

        // Set the pagination container style
        paginationContainer.style.display = "flex";
        paginationContainer.style.justifyContent = "space-between";
        paginationContainer.style.alignItems = "center";

        // Showing result count info (e.g., Showing 1 to 25 of 3556 results)
        let start = (currentPage - 1) * perpage + 1;
        let end = Math.min(currentPage * perpage, total);
        let showingInfo = Showing ${start} to ${end} of ${total} results;
        paginationContainer.insertAdjacentHTML('beforeend', showingInfo);

        // Create Pagination Buttons

        // << First button
        let firstButton = <li class="page-item ${currentPage === 1 ? 'disabled' : ''}" data-bs-toggle="tooltip" data-bs-placement="top" title="First Page"><button class=" page-link" onclick="loadThemes(1)" >«</button> </li>;
        
        // < Previous button
        let prevButton = <li class="page-item ${currentPage > 1 ? '' : 'disabled'}" data-bs-toggle="tooltip" data-bs-placement="top" title="Previous"><button class=" page-link" onclick="loadThemes(${currentPage - 1})" >‹</button> </li>;
        
        // Next button
        let nextButton = <li class="page-item ${currentPage < lastPage ? '' : 'disabled'}" data-bs-toggle="tooltip" data-bs-placement="top" title="Next"><button class="page-link" onclick="loadThemes(${currentPage + 1})" >›</button> </li>;
        
        // >> Last button
        let lastButton = <li class="page-item ${currentPage === lastPage ? 'disabled' : ''}" data-bs-toggle="tooltip" data-bs-placement="top" title="Last Page"><button class=" page-link" onclick="loadThemes(${lastPage})" >»</button> </li>;

        // Page Number Buttons (Dynamically show a range of pages around the current page)
        let pageButtons = '';
        let range = 2; // Show 2 pages before and after the current page
        let startPage = Math.max(1, currentPage - range);
        let endPage = Math.min(lastPage, currentPage + range);

        // Generate page numbers
        for (let i = startPage; i <= endPage; i++) {
            pageButtons += <li class="page-item ${i === currentPage ? 'active' : ''}"><button class="page-link " onclick="loadThemes(${i})">${i}</button> </li>;
        }

        // Add the pagination buttons and page numbers
        paginationContainer.insertAdjacentHTML('beforeend', 
            <nav aria-label="Page navigation example">
                <ul class="pagination">
                    ${firstButton}
                    ${prevButton}
                    ${pageButtons}
                    ${nextButton}
                    ${lastButton}
                </ul>
            </nav>
        );

        // Update perpage dropdown if changed
        document.getElementById('perpage').value = perpage;
    }

foreach($components as $comp){

    $amount = 0;

    // 👉 FIXED
    if($comp->calculation_type == 'fixed'){
        $amount = $comp->amount;
    }

    // 👉 PERCENTAGE
    elseif($comp->calculation_type == 'percentage'){
        $amount = ($basic * $comp->amount)/100;
    }

    // 👉 FORMULA
    elseif($comp->calculation_type == 'formula'){
        $amount = evalFormula($comp->formula, $basic);
    }

    // 👉 RULE BASED
    if($comp->rule_id){

        $rule = DB::table('egc_payroll_rules')->where('sno',$comp->rule_id)->first();

        $condition = json_decode($rule->condition_json,true);

        if($rule->rule_type == 'esi'){

            if($basic <= $condition['max_salary']){
                $amount = ($basic * $condition['employee_percent'])/100;
            }else{
                $amount = 0;
            }
        }

        if($rule->rule_type == 'pf'){

            $limit = min($basic, $condition['max_salary']);
            $amount = ($limit * $condition['employee_percent'])/100;
        }

    }

}

public function runPayroll(Request $req)
{
    $month = $req->month;
    $year = $req->year;
    $user_id = $req->user()->user_id;

    // 🔥 Get staff
    $staffs = DB::table('egc_staff')
        ->where('status',0)
        ->get();

    foreach($staffs as $staff){

        $basic = $staff->basic_salary;

        // 🔥 Attendance
        $attendance = DB::table('egc_attendance_summary')
            ->where('staff_id',$staff->sno)
            ->where('month',$month)
            ->where('year',$year)
            ->first();

        $present = $attendance->present_days ?? 0;
        $absent = $attendance->absent_days ?? 0;

        // 🔥 LOP Calculation
        $lop = ($basic / 30) * $absent;

        // 🔥 Get Staff Salary Structure
        $components = DB::table('egc_staff_salary_structure as s')
            ->join('egc_salary_components_new as c','c.sno','=','s.component_id')
            ->leftJoin('egc_payroll_rules as r','r.sno','=','c.rule_id')
            ->where('s.staff_id',$staff->sno)
            ->where('s.status',0)
            ->select(
                's.*',
                'c.component_name',
                'c.type',
                'c.calculation_type as master_calc',
                'c.rule_id',
                'r.rule_type',
                'r.condition_json'
            )
            ->get();

        $earnings = 0;
        $deductions = 0;

        // 🔥 Create payroll entry first
        $entry_id = DB::table('egc_payroll_entries')->insertGetId([
            'staff_id'=>$staff->sno,
            'company_id'=>$staff->company_id ?? 0,
            'payroll_month'=>$month,
            'payroll_year'=>$year,
            'basic_salary'=>$basic,
            'present_days'=>$present,
            'absent_days'=>$absent,
            'lop_amount'=>$lop,
            'created_at'=>now(),
            'created_by'=>$user_id
        ]);

        foreach($components as $comp){

            $amount = 0;

            // ✅ FIXED
            if($comp->calculation_type == 'fixed'){
                $amount = $comp->amount;
            }

            // ✅ PERCENTAGE
            elseif($comp->calculation_type == 'percentage'){
                $amount = ($basic * $comp->amount)/100;
            }

            // ✅ FORMULA
            elseif($comp->calculation_type == 'formula'){
                $amount = $this->evalFormula($comp->formula, $basic);
            }

            // 🔥 APPLY RULE ENGINE
            if($comp->rule_id){

                $condition = json_decode($comp->condition_json,true);

                // 👉 ESI
                if($comp->rule_type == 'esi'){
                    if($basic <= $condition['max_salary']){
                        $amount = ($basic * $condition['employee_percent'])/100;
                    }else{
                        $amount = 0;
                    }
                }

                // 👉 PF
                if($comp->rule_type == 'pf'){
                    $limit = min($basic, $condition['max_salary']);
                    $amount = ($limit * $condition['employee_percent'])/100;
                }
            }

            // 🔥 Add to totals
            if($comp->type == 'earning'){
                $earnings += $amount;
            }else{
                $deductions += $amount;
            }

            // 🔥 Save component
            DB::table('egc_payroll_entry_components')->insert([
                'payroll_entry_id'=>$entry_id,
                'component_id'=>$comp->component_id,
                'component_name'=>$comp->component_name,
                'type'=>$comp->type,
                'amount'=>$amount,
                'created_at'=>now()
            ]);

        }

        // 🔥 Add LOP to deduction
        $deductions += $lop;

        // 🔥 Final Net Salary
        $net = $earnings - $deductions;

        // 🔥 Update entry
        DB::table('egc_payroll_entries')
            ->where('sno',$entry_id)
            ->update([
                'total_earnings'=>$earnings,
                'total_deductions'=>$deductions,
                'net_salary'=>$net,
                'updated_at'=>now()
            ]);

    }

    return response()->json(['status'=>200,'message'=>'Payroll processed']);
}
🧠 3️⃣ FORMULA ENGINE
public function evalFormula($formula, $basic)
{
    $formula = str_replace('basic', $basic, $formula);

    try{
        return eval("return $formula;");
    }catch(\Throwable $e){
        return 0;
    }
}

public function getPayslip($entryId)
{
    $data = DB::table('egc_payroll_entries as e')
        ->join('egc_staff as s','s.sno','=','e.staff_id')
        ->leftJoin('egc_staff_bank_details as b','b.staff_id','=','s.sno')
        ->where('e.sno',$entryId)
        ->select(
            'e.*',
            's.staff_name as name',
            's.staff_id as staff_code',
            'b.bank_name',
            'b.bank_account_no as account_number'
        )
        ->first();

    // 🔥 GET COMPONENTS
    $components = DB::table('egc_payroll_entry_components')
        ->where('payroll_entry_id',$entryId)
        ->get();

    $earnings = [];
    $deductions = [];

    foreach($components as $comp){
        if($comp->type == 'earning'){
            $earnings[] = $comp;
        }else{
            $deductions[] = $comp;
        }
    }

    $data->earnings = $earnings;
    $data->deductions = $deductions;

    return $data;
}
